function evaluateDistance(intensity, distanceMeters) {
  const distanceSquared = distanceMeters * distanceMeters;
  return intensity / Math.max(distanceSquared, 0.0001);
}

为什么除以 π？因为“每个方向相同的辐射度”投影到整个半球后总权重正好是 π；不除会凭空创造能量。

反照率决定“留下多少颜色”；除以 π 决定“怎样在整个上半球分摊”。Lambert 是方向上均匀的理想漫反射，不是镜面高光。

出射辐射度 = 保留能量 / 圆周率
function lambertRadiance(irradiance, albedo) {
  const reflectedEnergy = irradiance * albedo;
  return reflectedEnergy / Math.PI;
}

先算灯给 A 多少，再用 Lambert 把 A 变成“小面积面光源”，最后算 A 给 B 多少。拆成三步能避免漏掉某一段距离或把两个余弦用反。

① Eₐ = I · max(0,nₐ·lₐ) / rₐ²
② Lₐ = ρₐ · Eₐ / π
③ Gₐᵦ = V · max(0,nₐ·ωₐᵦ) · max(0,nᵦ·ωᵦₐ) / rₐᵦ²
④ Eᵦ ≈ Lₐ · Aₐ · Gₐᵦ

第 ④ 步把 A 当成足够小的面片，因此是教学近似；真实面积较大时应在 A 上积分或抽样。

正在计算…
算出 灯到A 的方向、距离、A接收余弦
A收到 = 光强 × A接收余弦 / 距离平方
A发出 = A反照率 × A收到 / 圆周率

算出 A到B 的方向、距离
几何项 = 可见性 × A发射余弦 × B接收余弦 / 距离平方
B收到的一次反弹 = A发出 × A面积 × 几何项
function evaluateOneBounce(scene) {
  const toLight = subtract(scene.lightPosition, scene.bouncePosition);
  const lightDistanceSquared = dot(toLight, toLight);
  const lightDirection = normalize(toLight);
  const bounceReceives = Math.max(0, dot(scene.bounceNormal, lightDirection));
  const bounceIrradiance =
    scene.lightIntensity * bounceReceives / lightDistanceSquared;

  const bounceRadiance =
    scene.bounceAlbedo * bounceIrradiance / Math.PI;

  const bounceToReceiver =
    subtract(scene.receiverPosition, scene.bouncePosition);
  const transferDistanceSquared = dot(bounceToReceiver, bounceToReceiver);
  const outgoingDirection = normalize(bounceToReceiver);
  const emitterCosine = Math.max(0, dot(scene.bounceNormal, outgoingDirection));
  const receiverCosineValue = Math.max(
    0,
    dot(scene.receiverNormal, scale(outgoingDirection, -1))
  );
  const geometry =
    scene.visibility * emitterCosine * receiverCosineValue /
    transferDistanceSquared;

  return {
    bounceIrradiance,
    bounceRadiance,
    geometry,
    receiverIrradiance:
      bounceRadiance * scene.bounceArea * geometry
  };
}

每个样本都要除以“它被选中的概率 PDF”。偏爱高贡献方向可以降噪，但漏掉 PDF 会让结果系统性变亮或变暗。

把所有方向当成无法全部统计的选民。随机问 N 个方向，每票按“它有多难被抽中”校正；问得越多，平均值越稳定。

J = ∫ f(θ)dθ
Ĵ = (1/N) Σ [ f(θᵢ) / p(θᵢ) ]

这是便于画图的一维半圆实验；真实 GI 使用三维半球立体角 dω，思想相同。

正在计算…
总票数 = 0
重复 N 次:
  按所选策略抽一个方向 θ
  查询该方向的入射贡献 f(θ)
  查询该方向被抽中的概率 p(θ)
  总票数 += f(θ) / p(θ)
最终估计 = 总票数 / N
function estimateHemisphere(sampleCount, sampler, random) {
  const votes = [];
  for (let index = 0; index < sampleCount; index += 1) {
    const sample = sampleDirection(sampler, random);
    const contribution = teachingIncoming(sample.angle);
    votes.push(contribution / sample.pdf);
  }
  const estimate =
    votes.reduce((sum, vote) => sum + vote, 0) / sampleCount;
  return { estimate, votes };
}

function estimateFloorBounce(floorX, sampleCount, random, lightX) {
  let sum = 0;
  for (let index = 0; index < sampleCount; index += 1) {
    const wallY = 0.15 + random() * 2.7;
    sum += transferThroughWall(lightX, floorX, wallY);
  }
  return sum / sampleCount;
}
