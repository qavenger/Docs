Flux 不是“最终亮度”

它记录这个 texel 所在表面接到了多少灯光，并乘上表面颜色。后续接收点是否面对它、距离多远、第二段是否被挡住，都还没有计算。

distanceSquared = dot(delta, delta)
direction = delta / sqrt(distanceSquared)

receiverCosine = max(dot(receiver.normal, direction), 0)
vplCosine = max(dot(vpl.normal, -direction), 0)

contribution =
  visibility *
  vpl.flux *
  receiverCosine *
  vplCosine /
  max(distanceSquared, epsilon)
function singleVplContribution(receiver, vpl, visible) {
  const delta = subtract3(vpl.position, receiver.position);
  const distanceSquared = dot3(delta, delta);
  const direction = scale3(
    delta,
    1 / Math.sqrt(Math.max(distanceSquared, 1e-6))
  );

  const receiverCosine = Math.max(
    dot3(receiver.normal, direction),
    0
  );
  const vplCosine = Math.max(
    dot3(vpl.normal, scale3(direction, -1)),
    0
  );
  const visibility = visible ? 1 : 0;

  return scale3(
    vpl.flux,
    visibility *
      receiverCosine *
      vplCosine /
      Math.max(distanceSquared, 0.08)
  );
}

function estimateReceiver(
  receiver,
  pixelIndex,
  sampleCount,
  distribution,
  frame,
  useVisibility
) {
  const sum = [0, 0, 0];

  for (let sampleIndex = 0; sampleIndex < sampleCount; sampleIndex += 1) {
    const vplIndex = chooseVplIndex(
      pixelIndex,
      sampleIndex,
      sampleCount,
      distribution,
      frame
    );
    const vpl = labVpls[vplIndex];
    const visible = useVisibility
      ? secondaryVisible(receiver.position, vpl.position)
      : true;
    const contribution = singleVplContribution(
      receiver,
      vpl,
      visible
    );

    sum[0] += contribution[0];
    sum[1] += contribution[1];
    sum[2] += contribution[2];
  }

  return scale3(sum, LAB_DISPLAY_GAIN / sampleCount);
}

function edgeAwarePass(source, stepWidth) {
  const result = new Float32Array(source.length);
  const kernel = [1, 4, 6, 4, 1];

  for (let y = 0; y < LAB_HEIGHT; y += 1) {
    for (let x = 0; x < LAB_WIDTH; x += 1) {
      const center = y * LAB_WIDTH + x;
      let weightSum = 0;
      const sum = [0, 0, 0];

      for (let ky = -2; ky <= 2; ky += 1) {
        for (let kx = -2; kx <= 2; kx += 1) {
          const sampleX = clampInt(
            x + kx * stepWidth,
            0,
            LAB_WIDTH - 1
          );
          const sampleY = clampInt(
            y + ky * stepWidth,
            0,
            LAB_HEIGHT - 1
          );
          const sample = sampleY * LAB_WIDTH + sampleX;
          const spatialWeight = kernel[kx + 2] * kernel[ky + 2];
          const depthDelta = Math.abs(
            labGBuffer.depth[center] - labGBuffer.depth[sample]
          );
          const centerNormal = readRgb(labGBuffer.normal, center);
          const sampleNormal = readRgb(labGBuffer.normal, sample);
          const normalWeight = Math.pow(
            Math.max(dot3(centerNormal, sampleNormal), 0),
            16
          );
          const sameObject =
            labGBuffer.objectId[center] === labGBuffer.objectId[sample]
              ? 1
              : 0.015;
          const weight =
            spatialWeight *
            Math.exp(-depthDelta / 0.035) *
            normalWeight *
            sameObject;

          accumulateRgb(sum, source, sample, weight);
          weightSum += weight;
        }
      }

      storeRgb(result, center, scale3(sum, 1 / Math.max(weightSum, 1e-6)));
    }
  }

  return result;
}

function accumulateTemporal(current) {
  const output = new Float32Array(current.length);
  let rejected = 0;

  if (!labState.temporalEnabled) {
    output.set(current);
    labHistory.color.set(current);
    labHistory.depth.set(labGBuffer.depth);
    labHistory.objectId.set(labGBuffer.objectId);
    labHistory.valid = true;
    labHistory.frameCount = 1;
    return { output, rejected: LAB_PIXEL_COUNT };
  }

  for (let pixel = 0; pixel < LAB_PIXEL_COUNT; pixel += 1) {
    const depthChanged =
      Math.abs(
        labGBuffer.depth[pixel] - labHistory.depth[pixel]
      ) > 0.05;
    const objectChanged =
      labGBuffer.objectId[pixel] !== labHistory.objectId[pixel];
    const reject =
      !labHistory.valid ||
      (labState.historyRejection && (depthChanged || objectChanged));
    const currentWeight = reject ? 1 : labState.historyAlpha;

    if (reject) rejected += 1;
    for (let channel = 0; channel < 3; channel += 1) {
      const index = pixel * 3 + channel;
      output[index] =
        labHistory.color[index] * (1 - currentWeight) +
        current[index] * currentWeight;
    }
  }

  labHistory.color.set(output);
  labHistory.depth.set(labGBuffer.depth);
  labHistory.objectId.set(labGBuffer.objectId);
  labHistory.valid = true;
  labHistory.frameCount += 1;
  return { output, rejected };
}

固定条纹或规则花纹 相关采样误差 黄金角或固定序列跨像素同步，误差呈低频结构。 像素级 scramble、蓝噪排序、随机旋转；保持 PDF 正确。
相机/光源移动时闪烁 帧间方差 每帧抽到不同 VPL；64 仍只覆盖 512/16384 等很小比例。 时序积累 + motion vectors + rejection；动态光强变化要缩短 history。
几何边缘被抹开 滤波跨表面 普通 Gaussian 不知道两像素来自不同深度/法线。 bilateral / à-trous guidance，必要时按材质和 roughness 进一步拒绝。
墙后稳定发亮 模型偏差 基础 RSM 从未查询第二段 visibility；64、256、无限样本都会收敛到错误值。 显式次级 visibility：screen ray、SDF、voxel、ray query 或 ISM。
视锥外永远没有反弹 支持域偏差 RSM 只含光源相机可见的第一层表面。 扩大/分割光源视锥、多 RSM、点光 cubemap，或换用场景空间 GI 表示。

function renderRsmPass() {
  const position = lightPosition();
  const view = mat4LookAt(position, LIGHT_TARGET, [0, 1, 0]);
  const projection = mat4Perspective(
    LIGHT_FOV_RADIANS,
    1,
    LIGHT_NEAR,
    LIGHT_FAR
  );
  currentLightViewProjection = mat4Multiply(projection, view);

  gl.bindFramebuffer(gl.FRAMEBUFFER, rsmTarget.framebuffer);
  gl.drawBuffers([
    gl.COLOR_ATTACHMENT0,
    gl.COLOR_ATTACHMENT1,
    gl.COLOR_ATTACHMENT2
  ]);
  gl.viewport(0, 0, RSM_SIZE, RSM_SIZE);
  gl.clearColor(0, 0, 0, 0);
  gl.clear(gl.COLOR_BUFFER_BIT | gl.DEPTH_BUFFER_BIT);

  gl.useProgram(rsmProgram);
  bindMesh();
  gl.uniformMatrix4fv(
    gl.getUniformLocation(rsmProgram, "uViewProjection"),
    false,
    currentLightViewProjection
  );
  gl.uniform3fv(
    gl.getUniformLocation(rsmProgram, "uLightPosition"),
    position
  );
  gl.uniform1f(
    gl.getUniformLocation(rsmProgram, "uLightPower"),
    0.82
  );
  gl.enable(gl.POLYGON_OFFSET_FILL);
  gl.polygonOffset(1.5, 4.0);
  gl.drawArrays(gl.TRIANGLES, 0, meshVertexCount);
  gl.disable(gl.POLYGON_OFFSET_FILL);
  gl.bindFramebuffer(gl.FRAMEBUFFER, null);
  recordGlError("RSM MRT");
}
vec3 gatherIndirect(vec3 receiverPosition, vec3 receiverNormal) {
  vec4 lightClip =
    uLightViewProjection * vec4(receiverPosition, 1.0);
  vec2 centerUv =
    lightClip.xy / lightClip.w * 0.5 + 0.5;
  vec3 sum = vec3(0.0);

  for (int sampleIndex = 0; sampleIndex < 64; sampleIndex += 1) {
    if (sampleIndex >= uSampleCount) {
      break;
    }
    float fi = float(sampleIndex) + 0.5;
    float radius =
      sqrt(fi / float(uSampleCount)) * uSampleRadius;
    float angle = fi * 2.39996323;
    vec2 uv =
      centerUv + vec2(cos(angle), sin(angle)) * radius;

    if (
      any(lessThan(uv, vec2(0.0))) ||
      any(greaterThan(uv, vec2(1.0)))
    ) {
      continue;
    }

    vec4 encodedPosition = texture(uRsmPosition, uv);
    if (encodedPosition.a < 0.5) {
      continue;
    }
    vec3 vplPosition = decodePosition(encodedPosition);
    vec3 vplNormal = normalize(
      texture(uRsmNormal, uv).xyz * 2.0 - 1.0
    );
    vec3 delta = vplPosition - receiverPosition;
    float distanceSquared = dot(delta, delta);
    if (distanceSquared < 0.02) {
      continue;
    }

    vec3 direction = delta * inversesqrt(distanceSquared);
    float receiverCosine = max(
      dot(receiverNormal, direction),
      0.0
    );
    float vplCosine = max(
      dot(vplNormal, -direction),
      0.0
    );
    vec3 flux = texture(uRsmFlux, uv).rgb;
    sum +=
      flux *
      receiverCosine *
      vplCosine /
      max(distanceSquared, 0.08);
  }

  return sum * (
    uIndirectStrength *
    8.0 /
    float(uSampleCount)
  );
}
function secondaryVisible(receiverPosition, vplPosition) {
  if (receiverPosition[1] >= 1.049) {
    return true;
  }
  const direction = subtract3(vplPosition, receiverPosition);
  const minimum = [
    labState.objectOffset - 0.20,
    0.0,
    -0.17
  ];
  const maximum = [
    labState.objectOffset + 0.20,
    1.05,
    0.17
  ];

  const hit = segmentIntersectsBox(
    receiverPosition,
    direction,
    minimum,
    maximum
  );
  return !hit;
}

工程边界与继续阅读

RSM 是动态一次反弹估计器，不是完整路径追踪器。

BOUNDARIES

适合

  • 聚光灯、手电筒等有明确 light frustum 的动态局部光源。
  • 没有硬件光追、但可复用 shadow-map 光栅路径的 GPU。
  • 以一次漫反射为主，并接受低分辨率 gather + filtering 的场景。
  • 需要 Position / Normal / Flux 可视化调试的教学或工程原型。

不适合 / 需要混合

  • 大量动态灯：每盏 spot 一张 RSM，point light 常要六张。
  • 视锥外、光源看不见的反射面；RSM 支持域天然不完整。
  • 镜面、多次反弹、长距离跨房间传播。
  • 不能接受漏光时，必须补 visibility 或改用场景空间/射线方案。
