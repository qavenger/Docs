function directAt(position, normal, albedo, lightPosition) {
  const toLight = subtract(lightPosition, position);
  const distanceSquared = dot(toLight, toLight);
  const lightDirection = normalize(toLight);
  const receiverCosine = Math.max(0, dot(normal, lightDirection));
  const attenuation = 3.2 / (1 + 0.14 * distanceSquared);
  return albedo.map(
    (channel, index) =>
      channel * LIGHT_COLOR[index] * receiverCosine * attenuation
  );
}

function traceVisibilityAt(scene, pixelX, pixelY, options) {
  const rays = makeScreenRays(options.samples, options.seed);
  const outcomes = rays.map((ray) =>
    marchScreenRay(scene, pixelX, pixelY, ray, options.radius)
  );
  const hitCount = outcomes.filter(
    (outcome) => outcome.kind === 'hit'
  ).length;
  return {
    visibility: 1 - hitCount / options.samples,
    outcomes
  };
}

thickness 不是万能修复：太小会漏交点，太大会跨过真实间隙。离屏射线没有数据，增加 thickness 也不该把它伪造成 hit。

沿投影后的线走小步，每步把“射线应该在的深度”与深度纹理记录的“相机最先看到的深度”比较。两者足够接近时，暂时认为撞到了。

delta = rayViewZ - sceneViewZ
hit ⇔ 0 < delta < thickness

function marchSingleRay(scene, originX, originY, angle, options) {
  const directionX = Math.cos(angle);
  const directionY = -Math.sin(angle);
  const originIndex = originY * GRID_WIDTH + originX;
  const originDepth = scene.depth[originIndex];
  const steps = [];

  for (let step = 1; step <= options.maxSteps; step += 1) {
    const travel = step * options.stepPixels;
    const x = Math.round(originX + directionX * travel);
    const y = Math.round(originY + directionY * travel);
    if (x < 0 || x >= GRID_WIDTH || y < 0 || y >= GRID_HEIGHT) {
      return { kind: 'offscreen', step, steps };
    }
    const rayDepth = originDepth - step * options.depthSlope;
    const sceneDepth = scene.depth[y * GRID_WIDTH + x];
    const delta = rayDepth - sceneDepth;
    steps.push({ x, y, rayDepth, sceneDepth, delta });
    if (delta > 0 && delta < options.thickness) {
      return { kind: 'hit', x, y, step, steps };
    }
  }
  return { kind: 'miss', steps };
}

function bilateralFilter(raw, scene, radius) {
  const output = new Float32Array(raw.length);
  forEachPixel((x, y, index) => {
    let weightSum = 0;
    forEachNeighbor(x, y, radius, (neighborIndex, dx, dy) => {
      const depthDifference =
        Math.abs(scene.depth[index] - scene.depth[neighborIndex]);
      const normalSimilarity =
        Math.max(0, normalDot(scene, index, neighborIndex));
      const spatialWeight = Math.exp(-(dx * dx + dy * dy) * 0.32);
      const weight =
        spatialWeight *
        Math.exp(-depthDifference * 5.5) *
        Math.pow(normalSimilarity, 12);
      addWeightedRgb(output, index, raw, neighborIndex, weight);
      weightSum += weight;
    });
    divideRgb(output, index, Math.max(weightSum, 0.0001));
  });
  return output;
}

function accumulateFrame(current, scene, previous, options) {
  const output = new Float32Array(current.length);
  const validMask = new Float32Array(PIXEL_COUNT);
  forEachPixel((x, y, index) => {
    const previousX = x - scene.motionX[index];
    const previousIndex = pixelIndex(previousX, y);
    const valid =
      !options.cameraCut &&
      previousIndex >= 0 &&
      Math.abs(scene.depth[index] - previous.depth[previousIndex]) < 0.28 &&
      normalDotAcross(
        scene.normal,
        index,
        previous.normal,
        previousIndex
      ) > 0.78;
    validMask[index] = valid ? 1 : 0;
    mixRgb(
      output,
      index,
      current,
      index,
      previous.color,
      previousIndex,
      valid ? options.historyWeight : 0
    );
  });
  return { color: output, validMask };
}

AO 是“环境可见比例”，SSGI 是“命中表面带回的有色光”。教学合成把 Direct 乘 AO 后再加 GI；实际引擎应根据光照分层，避免把 AO 错乘到不该遮蔽的项上。

function compositeFrame(scene, ao, gi) {
  const color = new Float32Array(PIXEL_COUNT * 3);
  forEachPixel((x, y, index) => {
    for (let channel = 0; channel < 3; channel += 1) {
      const hdr =
        scene.direct[index * 3 + channel] * ao[index] +
        gi[index * 3 + channel];
      color[index * 3 + channel] = toneMap(hdr);
    }
  });
  return color;
}
07

Screen-space 的硬边界：滤波能降噪，不能创造没进屏幕的数据

把“估计器噪声”和“信息根本不存在”分开，才能选对补救方案。

FAILURE BOUNDARY
离屏缺失光源或反射面离开视口后，其 Depth/Normal/Radiance 不存在。空间与时序滤波都无法得到可靠新信息。
背面与第二层缺失GBuffer 只保存相机首先看见的一层；薄物体后方、遮挡后表面和背面容易漏光或断裂。
高速变光的历史滞后深度与法线仍匹配时，历史验证可能接受过期照明。高速运动光源应降低 α、加入光照变化检测，或依赖不需要历史的方案。
本页 Reference 的边界：64-ray Reference 仍然只看当前屏幕，因此它是“同一屏幕空间估计器的高样本参考”，不是路径追踪真值。要评估离屏漏光，必须另接场景空间或光追参考。
