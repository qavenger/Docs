把 probe 想成站在房间里转一圈的照度计：ray 命中哪一面，就记录那一方向看到的颜色；同时记录“多远才碰到东西”，供查询阶段判断墙体。

function traceProbeRay(origin, angle, light) {
  const direction = [Math.cos(angle), Math.sin(angle)];
  const hit = nearestSceneHit(origin, direction);
  if (!hit) return { radiance: 0.025, distance: 2.4 };

  const toLight = subtract2(light, hit.position);
  const lightDistance2 = dot2(toLight, toLight);
  const lightDirection = normalize2(toLight);
  const receiverCosine = Math.max(0, dot2(hit.normal, lightDirection));
  const visible = !segmentHitsOccluder(hit.position, light);
  const radiance = visible
    ? 1.8 * hit.albedo * receiverCosine / (1 + 1.4 * lightDistance2)
    : 0.008;
  return { radiance, distance: hit.distance };
}

两个 atlas：颜色与几何证据分开存

方向格子一一对应；上排是 irradiance，下排是 distance mean / second moment。

高亮格显示当前 direction 的 texel 与 moments。

颜色 atlas 回答“从这个方向来的光有多少”；distance atlas 回答“这个方向多远有墙，以及测量有多稳定”。两者职责不同，不能只存颜色。

texel = directionToAtlas(ray.direction)
irradiance[texel] += ray.radiance * cosineKernel
distanceMean[texel] += ray.distance
distanceSecondMoment[texel] += ray.distance * ray.distance
function rebuildAtlas(samples, texelCount) {
  const atlas = Array.from({ length: texelCount }, () => ({
    irradiance: 0,
    mean: 0,
    secondMoment: 0,
    count: 0
  }));
  for (const sample of samples) {
    const texel = directionToAtlas(sample.angle, texelCount);
    const cell = atlas[texel];
    cell.irradiance += sample.radiance;
    cell.mean += sample.distance;
    cell.secondMoment += sample.distance * sample.distance;
    cell.count += 1;
  }
  return atlas.map(normalizeAtlasCell);
}

function gatherTeachingCell(queryX, useVisibility) {
  let weighted = 0;
  let weightSum = 0;
  for (const probe of gatherProbes) {
    const trilinear = Math.max(0, 1 - Math.abs(queryX - probe.x));
    const crossesWall = queryX * probe.x < 0;
    const visibility = useVisibility && crossesWall ? 0.025 : 1;
    const weight = trilinear * visibility * probe.validity;
    weighted += weight * probe.irradiance;
    weightSum += weight;
  }
  return weighted / Math.max(weightSum, 1e-5);
}

线段是否穿过本 Demo 的墙

bool SegmentBlockedByDemoWall(float3 a, float3 b) {
  if (a.x * b.x >= 0.0) return false; // 同侧
  float t = -a.x / (b.x - a.x);       // x=0 交点
  float3 hit = lerp(a, b, t);
  return hit.y > -0.78 && hit.y < 0.68 &&
         abs(hit.z) < 0.95;
}

float FinalWeight(float trilinearWeight,
                  bool wallBlocked,
                  float validity) {
  float visibility = wallBlocked ? 0.02 : 1.0;
  return trilinearWeight * visibility * validity;
}

Advantages / Limits

Probe 省下的是密集空间求解，不是所有困难

它把昂贵照明采样摊到稀疏位置，再依靠重建覆盖像素。质量取决于“场是否足够低频”与 visibility 是否可靠。

优势

查询成本与场景三角形数弱相关；适合大空间 diffuse GI；Probe 数据可流式、级联或分 brick；DDGI 可支持动态灯和一定程度动态几何。

空间限制

墙、门缝、角落与小尺度遮挡不是平滑场。网格过稀时，再复杂的插值也不能恢复未采样信息。

时间限制

Baked 不响应任意动态灯；DDGI 要分批发 ray 并历史融合，快速变化会有延迟、拖影或更新预算尖峰。

与 PRT Probe 的分界

Irradiance Probe 存某个光照状态下的结果；PRT 存固定几何对可变入射光的传输算子。两者都可能用 SH，但灯变化时一个需要重算/更新结果，另一个在适用假设内做系数点积。继续阅读：PRT / Transport Probes →
