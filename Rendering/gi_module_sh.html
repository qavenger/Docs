function projectEnvironment(preset, yaw, pitch, sampleCount) {
  const coefficients = Array.from(
    { length: 9 },
    () => [0, 0, 0]
  );
  const weight = 4 * Math.PI / sampleCount;

  for (let s = 0; s < sampleCount; s += 1) {
    const direction = fibonacciDirection(s, sampleCount);
    const radiance = sampleEnvironment(
      preset,
      direction,
      yaw,
      pitch
    );
    const Y = basis9(direction);

    for (let i = 0; i < 9; i += 1) {
      for (let channel = 0; channel < 3; channel += 1) {
        coefficients[i][channel] +=
          radiance[channel] * Y[i] * weight;
      }
    }
  }
  return coefficients;
}

实验 2：重新投影环境，还是旋转已有系数？

两条路径应该得到同一结果，但运行时成本完全不同。路径 A 重新访问环境并做球面积分；路径 B 只在每个 band 内混合已有系数。为了让算法可见，本页通过方向采样拟合旋转矩阵 D；生产代码通常使用解析 SH rotation、递推或经过验证的库。

function buildBandRotation(yaw, pitch, fitSamples = 96) {
  const ranges = [[0, 1], [1, 4], [4, 9]];
  const full = makeMatrix(9, 9, 0);
  const bands = [];

  for (const [begin, end] of ranges) {
    const width = end - begin;
    const G = makeMatrix(width, width, 0);
    const C = makeMatrix(width, width, 0);

    for (let s = 0; s < fitSamples; s += 1) {
      const local = fibonacciDirection(s, fitSamples);
      const world = rotateWorld(local, yaw, pitch);
      const a = basis9(world).slice(begin, end);
      const b = basis9(local).slice(begin, end);
      addOuterProduct(C, a, b);
      addOuterProduct(G, b, b);
    }

    const D = multiplyMatrices(C, invertMatrix(G));
    bands.push(D);
    for (let row = 0; row < width; row += 1) {
      for (let column = 0; column < width; column += 1) {
        full[begin + row][begin + column] = D[row][column];
      }
    }
  }
  return { full, bands };
}

实验 3：L0 为什么完全不动？

L0 的基函数在球面每个方向都是同一个常数 0.282095。环境转动只是把相同的一组亮度搬到别的方向；球面总平均不会变。

function applyCoefficientRotation(base, D) {
  const output = Array.from(
    { length: 9 },
    () => [0, 0, 0]
  );

  for (let row = 0; row < 9; row += 1) {
    for (let column = 0; column < 9; column += 1) {
      for (let channel = 0; channel < 3; channel += 1) {
        output[row][channel] +=
          D[row][column] * base[column][channel];
      }
    }
  }
  return output;
}

function dot3(a, b) {

function l1Data() {
  const base = l1VectorFromCoefficients(baseCoefficients);
  const expected = rotateWorld(base, state.yaw, state.pitch);
  const actual =
    l1VectorFromCoefficients(rotatedCoefficients);
  const baseLength = Math.max(length3(base), 1e-12);
  const expectedLength = Math.max(
    length3(expected),
    1e-12
  );
  const actualLength = Math.max(
    length3(actual),
    1e-12
  );
  const cosine = Math.max(
    -1,
    Math.min(
      1,
      dot3(expected, actual) /
        (expectedLength * actualLength)
    )
  );
  return {
    base,
    expected,
    actual,
    angle: Math.acos(cosine) / DEG,
    lengthDrift:
      Math.abs(actualLength - baseLength) / baseLength
  };
}

实验 5：L2 的“五维”到底是什么意思？

五维不是说场景多了两个空间维度，而是同一组二阶方向形状需要五个独立数字。你可以把这五个数原样看成 5D 向量；也可以把它们重新包装成一个 3×3 对称、迹为零的矩阵 Q。环境旋转时，Q′=RQRᵀ，再转回五个系数。该路径与数值 D₂c 是第二次独立验证。

function l2CoefficientsToTensor(c) {
  const K = 1.092548;
  const C = 0.315392;
  const D = 0.546274;
  return [
    [-C * c[2] + D * c[4], K * c[0] / 2, K * c[3] / 2],
    [K * c[0] / 2, -C * c[2] - D * c[4], K * c[1] / 2],
    [K * c[3] / 2, K * c[1] / 2, 2 * C * c[2]]
  ];
}

function tensorToL2Coefficients(Q) {
  const K = 1.092548;
  const C = 0.315392;
  const D = 0.546274;
  return [
    2 * Q[0][1] / K,
    2 * Q[1][2] / K,
    Q[2][2] / (2 * C),
    2 * Q[0][2] / K,
    (Q[0][0] - Q[1][1]) / (2 * D)
  ];
}

function rotationMatrix3(yaw, pitch) {
  const x = rotateWorld([1, 0, 0], yaw, pitch);
  const y = rotateWorld([0, 1, 0], yaw, pitch);
  const z = rotateWorld([0, 0, 1], yaw, pitch);
  return [
    [x[0], y[0], z[0]],
    [x[1], y[1], z[1]],
    [x[2], y[2], z[2]]
  ];
}

function rotateTensor(Q, R) {
  return multiplyMatrices(
    multiplyMatrices(R, Q),
    transposeMatrix(R)
  );
}

实验 6：Band energy 是旋转的“单元测试”

正交旋转只会在同一 band 内重新分配系数，不会改变这一 band 的平方和。若能量明显改变，通常是混了 convention、矩阵方向写反、把不同 band 混在一起，或数值积分样本过少。

旋转像在同一只盒子里重新摇匀五颗 L2 砝码：每颗读数会变，但五颗读数的“平方总重量”不变。L0、L1、L2 是三只不同盒子，不能互相倒。

function bandEnergies(coefficients) {
  const ranges = [[0, 1], [1, 4], [4, 9]];
  return ranges.map(([begin, end]) => {
    let energy = 0;
    for (let i = begin; i < end; i += 1) {
      for (let channel = 0; channel < 3; channel += 1) {
        energy += coefficients[i][channel] ** 2;
      }
    }
    return energy;
  });
}

实验 7：Diffuse convolution 为什么让 SH 特别适合漫反射？

漫反射 irradiance 要把整个半球的入射光乘 max(n·ω,0) 再积分。对于 SH，这个卷积不必再次遍历所有方向：前三个 band 分别乘固定常数 π、2π/3、π/4。高 band 衰减更多，所以尖锐细节自然变宽。

  float y = d.y;

int activeCount() {
  return uOrder == 0 ? 1 : uOrder == 1 ? 4 : 9;
}

vec3 diffuseIrradiance(vec3 normal) {
  vec3 irradiance = vec3(0.0);
  for (int i = 0; i < 9; i += 1) {
    if (i >= activeCount()) {
      break;
    }
    float bandScale =
      i == 0 ? 3.14159265 :
      i < 4 ? 2.09439510 :
                0.78539816;
    irradiance +=
      uDirect[i] * shBasis(i, normal) * bandScale;
  }
  return max(irradiance, vec3(0.0));
}

实验 8：样本加到 4096，为什么尖窗仍然有瑕疵？

function windowedReconstruction(
  coefficients,
  direction,
  strength
) {
  const Y = basis9(direction);
  const output = [0, 0, 0];

  for (let i = 0; i < 9; i += 1) {
    const band = i === 0 ? 0 : i < 4 ? 1 : 2;
    const hann =
      0.5 * (1 + Math.cos(Math.PI * band / 3));
    const weight = 1 + (hann - 1) * strength;
    for (let channel = 0; channel < 3; channel += 1) {
      output[channel] +=
        weight * coefficients[i][channel] * Y[i];
    }
  }
  return output;
}

把 SH 放回 GI：Probe 与 PRT 分别在存什么？

SH 只是“方向函数的压缩格式”。它可以装在 Irradiance/Radiance Probe 里，也可以作为 PRT 的 lighting/transfer 基；这两条管线的物理量和更新方式不同。

方向光照环境贴图、天空、多盏低频化光源
投影 / 旋转得到当前 lighting SH；旋转环境时可旋转系数
空间或表面数据Probe 存某位置的 radiance/irradiance；PRT 存接收点 transfer
运行时组合Probe 插值后按法线求值；PRT 做 lighting · transfer

Irradiance / Radiance Probe

空间采样点携带一组 SH。相机/物体移动时插值邻近 Probe，再用表面法线求值。难点通常是空间泄漏、可见性和动态更新，不是 9 次乘加。

function fibonacciDirection(index, count) {
  const y = 1 - 2 * (index + 0.5) / count;
  const radius = Math.sqrt(Math.max(0, 1 - y * y));
  const goldenAngle = Math.PI * (3 - Math.sqrt(5));
  const azimuth = goldenAngle * index;
  return [
    Math.cos(azimuth) * radius,
    y,
    Math.sin(azimuth) * radius
  ];
}

function basis9(direction) {
  const [x, y, z] = direction;
  return [
    0.282095,
    0.488603 * y,
    0.488603 * z,
    0.488603 * x,
    1.092548 * x * y,
    1.092548 * y * z,
    0.315392 * (3 * z * z - 1),
    1.092548 * x * z,
    0.546274 * (x * x - y * y)
  ];
}

展开闭合实现

function rotateWorld(v, yaw, pitch) {
  const cp = Math.cos(pitch);
  const sp = Math.sin(pitch);
  const afterPitch = [
    v[0],
    cp * v[1] - sp * v[2],
    sp * v[1] + cp * v[2]
  ];
  const cy = Math.cos(yaw);
  const sy = Math.sin(yaw);
  return [
    cy * afterPitch[0] + sy * afterPitch[2],
    afterPitch[1],
    -sy * afterPitch[0] + cy * afterPitch[2]
  ];
}

function rotateLocal(v, yaw, pitch) {
  const cy = Math.cos(-yaw);
  const sy = Math.sin(-yaw);
  const afterYaw = [
    cy * v[0] + sy * v[2],
    v[1],
    -sy * v[0] + cy * v[2]
  ];
  const cp = Math.cos(-pitch);
  const sp = Math.sin(-pitch);
  return [
    afterYaw[0],
    cp * afterYaw[1] - sp * afterYaw[2],
    sp * afterYaw[1] + cp * afterYaw[2]
  ];
}

function dot3(a, b) {
  return a[0] * b[0] +
    a[1] * b[1] +
    a[2] * b[2];
}

function length3(v) {
  return Math.hypot(v[0], v[1], v[2]);
}

function normalize3(v) {
  const length = Math.max(length3(v), 1e-12);
  return [
    v[0] / length,
    v[1] / length,
    v[2] / length
  ];
}

function sphericalLobe(direction, axis, sharpness) {
  const a = normalize3(axis);
  return Math.exp(
    (dot3(direction, a) - 1) * sharpness
  );
}

function makeMatrix(rows, columns, value = 0) {
  return Array.from(
    { length: rows },
    () => Array(columns).fill(value)
  );
}

function addOuterProduct(matrix, column, row) {
  for (let y = 0; y < matrix.length; y += 1) {
    for (let x = 0; x < matrix[y].length; x += 1) {
      matrix[y][x] += column[y] * row[x];
    }
  }
}

function multiplyMatrices(a, b) {
  const output = makeMatrix(
    a.length,
    b[0].length,
    0
  );
  for (let row = 0; row < a.length; row += 1) {
    for (
      let column = 0;
      column < b[0].length;
      column += 1
    ) {
      for (let k = 0; k < b.length; k += 1) {
        output[row][column] +=
          a[row][k] * b[k][column];
      }
    }
  }
  return output;
}

function transposeMatrix(matrix) {
  return matrix[0].map(
    (_, column) => matrix.map(
      (row) => row[column]
    )
  );
}

function invertMatrix(matrix) {
  const n = matrix.length;
  const augmented = matrix.map((row, y) => [
    ...row,
    ...Array.from(
      { length: n },
      (_, x) => x === y ? 1 : 0
    )
  ]);

  for (let column = 0; column < n; column += 1) {
    let pivot = column;
    for (let row = column + 1; row < n; row += 1) {
      if (
        Math.abs(augmented[row][column]) >
        Math.abs(augmented[pivot][column])
      ) {
        pivot = row;
      }
    }
    if (Math.abs(augmented[pivot][column]) < 1e-12) {
      throw new Error('SH band fit matrix is singular');
    }
    [augmented[column], augmented[pivot]] =
      [augmented[pivot], augmented[column]];
    const divisor = augmented[column][column];
    for (let x = 0; x < 2 * n; x += 1) {
      augmented[column][x] /= divisor;
    }
    for (let row = 0; row < n; row += 1) {
      if (row === column) continue;
      const factor = augmented[row][column];
      for (let x = 0; x < 2 * n; x += 1) {
        augmented[row][x] -=
          factor * augmented[column][x];
      }
    }
  }
  return augmented.map((row) => row.slice(n));
}
