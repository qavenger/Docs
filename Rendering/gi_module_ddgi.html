function traceProbeRay(origin, direction) {
  let distance = 0.35;
  for (let step = 0; step < 64 && distance < MAX_TRACE; step += 1) {
    const point = add(origin, scale(direction, distance));
    const scene = sampleScene(point.x, point.y);
    if (scene.distance < HIT_EPSILON) {
      const normal = estimateNormal(point);
      return {
        radiance: shadeHit(point, normal, scene.material),
        distance,
        hit: true,
      };
    }
    distance += clamp(scene.distance * 0.78, 0.18, 2.2);
  }
  return { radiance: SKY_RADIANCE, distance: MAX_TRACE, hit: false };
}

每个小 Tile 是一个 Probe；Tile 内从左到右对应本页二维方向。教学版直接显示 ray radiance，生产更新还会过滤/卷积成 irradiance texel。

球面不能无缝摊成矩形。八面体映射先把方向投到八面体，再把背面折回菱形。它像世界地图投影：坐标更好存，但边界采样要格外小心。

p = n / (|n.x| + |n.y| + |n.z|)
p.z < 0 时，把 p.xy 沿菱形边折回

—

归一化方向
除以三个分量绝对值之和
如果落在八面体背面:
  将二维坐标折回正面菱形
把 [-1, 1] 映射到 Atlas tile 的 [0, 1]
function octEncode(normal) {
  const sum = Math.abs(normal.x) + Math.abs(normal.y) + Math.abs(normal.z);
  let x = normal.x / sum;
  let y = normal.y / sum;
  const z = normal.z / sum;
  if (z < 0) {
    const oldX = x;
    x = (1 - Math.abs(y)) * Math.sign(oldX || 1);
    y = (1 - Math.abs(oldX)) * Math.sign(y || 1);
  }
  return { u: x * 0.5 + 0.5, v: y * 0.5 + 0.5 };
}
STAGE 03 · DISTANCE MOMENTS

平均距离不够：还要知道距离有多不稳定

一阶矩是平均距离，二阶矩用于估计方差。着色点比平均遮挡距离更远时，Chebyshev 权重会降低该 Probe 的可信度，从而抑制穿墙漏光。

03

距离一阶矩 / 方差 / 可见性曲线

移动查询距离，观察可见性从 1 降到较小值。

独立 Demo D–E

Distance moments Atlas

Tile 上半部是 mean，下半部是由二阶矩导出的 standard deviation；亮度分别按最大追踪距离与显示范围归一化。

Chebyshev visibility

—

Probe 报告“这个方向通常 5 米撞墙，波动约 1 米”。如果当前表面离它 12 米，就很可能在墙后；如果只离 3 米，通常仍可见。方差让判断不是生硬的 0/1。

variance = max(E[d²] − E[d]², 0.12)
p = distance ≤ mean ? 1 : variance / (variance + (distance − mean)²)
visibility = clamp((p − 0.08) / 0.92, 0.03, 1)

—

读取该方向的平均命中距离 mean
读取平均距离平方 secondMoment
variance = max(secondMoment - mean * mean, 最小方差)
如果查询点比 mean 更近:
  可见性 = 1
否则:
  用 Chebyshev 上界平滑降低可信度
function visibilityWeight(queryDistance, mean, secondMoment) {
  const variance = Math.max(secondMoment - mean * mean, 0.12);
  if (queryDistance <= mean) return 1;
  const delta = queryDistance - mean;
  const probability = variance / (variance + delta * delta);
  return clamp((probability - 0.08) / 0.92, 0.03, 1);
}
STAGE 04 · TEMPORAL UPDATE

Hysteresis 能降噪，也会把旧光拖进新帧

“当前帧”与“历史”按滞后系数混合。硬切光源后连续点推进帧，可以直接看到高 hysteresis 收敛更慢；重置历史会立即响应但噪声更大。

0.92 表示“保留 92% 旧记忆，只接受 8% 新观察”。稳定场景很平滑；光突然移动时，旧记忆会拖尾。它不是免费的降噪。

historynew = h · historyold + (1 − h) · current

—

function updateHistory(previous, current, hysteresis, hardReset) {
  if (hardReset || !Number.isFinite(previous)) return current;
  return previous * hysteresis + current * (1 - hysteresis);
}

for 每个 Probe:
  计算双线性位置权重
  计算 Probe 是否位于表面正面

function gatherIrradiance(point, normal, includeLedger = false) {
  const neighbors = gatherNeighbors(point);
  let weighted = [0, 0, 0];
  let weightSum = 0;
  const ledger = [];

  for (const item of neighbors) {
    const toPoint = subtract(point, item.probe.position);
    const distance = length(toPoint);
    const direction = normalize(toPoint);
    const moments = sampleMoments(item.probe, direction);
    const visibility = state.visibility
      ? visibilityWeight(distance, moments.mean, moments.second)
      : 1;
    const facing = clamp((dot(normal, scale(direction, -1)) + 0.25) / 1.25, 0.05, 1);
    const weight = item.bilinear * visibility * facing * item.probe.stateWeight;
    weighted = add3(weighted, scale3(item.probe.irradiance, weight));
    weightSum += weight;
    if (includeLedger) {
      ledger.push({
        probe: item.probe,
        bilinear: item.bilinear,
        visibility,
        facing,
        weight,
        moments,
        distance,
      });
    }
  }
  return {
    value: scale3(weighted, 1 / Math.max(weightSum, 0.0001)),
    weightSum,
    ledger,
  };
}
STAGE 06 · RELOCATION & CLASSIFICATION

Probe 生在墙里时，不要让它继续污染周围

重定位沿 SDF 梯度把 Probe 推到自由空间；若推不出去或附近几乎全是背面命中，就标为 inactive。下面的开关直接改变实际 Probe 位置与 Gather 权重。

06

原始格点 / 重定位后 / inactive

红叉为墙内或无效 Probe，黄箭头为实际位移。

独立 Demo H
—

规则格点不知道墙在哪里。SDF 梯度像“最近出口”的箭头；沿它把 Probe 推出墙。若出口太远，则让 Probe 暂停贡献，比写入错误的强光更安全。

offset = normalSDF · clamp(threshold − signedDistance, 0, maxOffset)
读取 Probe 原始位置的 signed distance
如果距离小于安全阈值:
  用 SDF 中心差分估计离开表面的方向
  沿方向推离，但不超过最大偏移
如果新位置仍在墙内:
  标记 inactive，不参与 Gather
function relocateProbe(basePosition) {
  const signedDistance = signedSceneDistance(basePosition.x, basePosition.y);
  if (!state.relocation || signedDistance >= state.relocateThreshold) {
    return { position: basePosition, active: signedDistance > 0.05, offset: 0 };
  }
  const gradient = estimateSdfGradient(basePosition);
  const offset = clamp(
    state.relocateThreshold - signedDistance,
    0,
    state.relocateMax,
  );
  const position = add(basePosition, scale(gradient, offset));
  return {
    position,
    active: signedSceneDistance(position.x, position.y) > 0.05,
    offset,
  };
}

STAGE 09 · DECISION TABLE

DDGI 的优势、代价与高速光源结论

动态不等于无历史；可扩展也不等于每个变化都即时收敛。

维度优势劣势 / 风险本页观察入口
动态几何/光源Probe 可持续更新，不必整场烘焙。每帧只发有限射线，依赖历史时会滞后。Stage 04 硬切历史。
遮挡距离矩能给穿墙路径降权。矩是统计近似，薄墙、方差与偏置仍会漏光。Stage 03/05 关闭 visibility 对照。
覆盖范围滚动体积能围绕相机复用大部分 Probe。新进入区域没有历史，会短暂欠收敛。Stage 07 reused/new。
高速大量运动光源提高射线/Probe 更新率后可响应。默认高 hysteresis 不适合“立即正确”；成本随变化区域增加。Stage 04 把 h 设到 0.98。
无时序要求可把 h=0 做 single-frame 更新。噪声和射线成本显著增加；本课程无时序首选仍是 Voxel GI / RCGI 教学路径。Stage 04 把 h 设为 0。
