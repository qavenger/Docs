// FFT bin id → 物理波向量 k (rad/m)
float2 kFromId(uint2 id) {
    int n = (int)id.x - (int)N / 2;
    int m = (int)id.y - (int)N / 2;
    return float2(n, m) * (6.2831853 / L_phys);
}
━ 辅助函数 ━
// 复数乘法 (a + ib)·(c + id)
float2 cmul(float2 a, float2 b) {
    return float2(a.x*b.x - a.y*b.y, a.x*b.y + a.y*b.x);
}

// 均匀随机 → 用于 Box-Muller
float2 Hash22(uint2 p) {
    uint n = p.x * 73856093u ^ p.y * 19349663u;
    n = (n << 13) ^ n;
    n = n * (n * n * 15731u + 789221u) + 1376312589u;
    float u1 = frac((float)n * 2.3283064e-10);
    n = n * 747796405u + 2891336453u;
    float u2 = frac((float)n * 2.3283064e-10);
    return float2(max(u1, 1e-7), u2);
}

// Phillips 谱 P(k)：解析函数，被 H₀ pass 内联调用（不独立 dispatch）
float Phillips(float2 k) {
    float k2 = dot(k, k);
    if (k2 < 1e-12) return 0.0;
    float2 windDir = float2(cos(theta), sin(theta));
    float L_wave = V * V / 9.81;
    float kw   = max(0.0, dot(normalize(k), windDir));
    float damp = exp(-k2 * (L_wave*1e-3)*(L_wave*1e-3));
    return A * exp(-1.0/(k2*L_wave*L_wave)) / (k2*k2) * kw*kw * damp;
}

Tessendorf FFT 海面 · 完整流程

9 步流水线：高斯噪声 → Phillips 谱 → H₀ → 时变 H(k,t) → 5× IFFT → 高度 / 法线 / Choppy 位移 → 着色合成。

每步右侧给出原理公式 + 核心 HLSL（点击展开）+ 实时中间结果预览。带 LIVE 标记的画布每帧重算。

01 高斯噪声 ξ（Box-Muller） SETUP
为每个 FFT bin 抽两个独立的 N(0,1) 高斯样本 ξr 与 ξi。后面所有"随机性"都来自这一步。 参数变化（风/L/A）时复用这两张噪声，只在点 Regen ξ 时重抽。
ξᵣ = √(-2 ln u₁) · cos(2π u₂)
ξᵢ = √(-2 ln u₁) · sin(2π u₂)   // u₁,u₂ ~ U(0,1)，输出 ξr,ξi 互相独立 ~ N(0,1)
HLSL · Box-Muller 离线烘焙 
// CPU 端（或一次性 CS）烘焙噪声贴图，运行期复用
RWTexture2D<float4> GaussianTex;       // .xy = (ξr, ξi)，.zw 备用

// 简化 PRNG（实际项目可用 Hash 或预生成）
float2 Hash22(uint2 p) {
    uint n = p.x * 73856093u ^ p.y * 19349663u;
    n = (n << 13) ^ n;
    n = n * (n * n * 15731u + 789221u) + 1376312589u;
    float u1 = frac(float(n) * 2.3283064e-10);
    n = n * 747796405u + 2891336453u;
    float u2 = frac(float(n) * 2.3283064e-10);
    return float2(max(u1, 1e-7), u2);
}

[numthreads(8, 8, 1)]
void CS_BakeGaussian(uint3 id : SV_DispatchThreadID) {
    float2 u = Hash22(id.xy);
    float  r = sqrt(-2.0 * log(u.x));
    float  a = 6.2831853 * u.y;
    GaussianTex[id.xy] = float4(r * cos(a), r * sin(a), 0, 0);
}
ξr N(0,1) 实部
ξi N(0,1) 虚部
02 Phillips 振幅谱 √P(k) FREQ
把"哪些频率上有多少能量"按风向加权地写到频域。单向方向因子 max(0, k̂·ŵ)² 只在沿风向那一侧给能量，逆风侧砍为 0 —— 这是 step 9 海面"沿风向流动"的关键。Step 4 的 +/- 配对会让 -k 侧也长出一个共轭瓣（视觉上仍是两瓣），但它和 +k 是同一束波的复共轭，不是独立的反向波。
P(k) = A · exp(-1/(k2L2)) / k4 · max(0, k̂ · ŵ)2 · exp(-k2ℓ2)  // 单向 cos²（不是 |·|²）
HLSL · Phillips 函数 
cbuffer SpectrumParams : register(b0) {
    uint   N;          float L_phys;
    float  V;          float2 windDir;
    float  A;
};

float Phillips(float2 k) {
    float k2 = dot(k, k);
    if (k2 < 1e-12) return 0.0;
    float L  = V * V / 9.81;
    float k4 = k2 * k2;
    float kw = dot(normalize(k), windDir);
    kw = max(0.0, kw);                            // 单向：只保留沿风向能量（→ 有方向感）
    float damp = exp(-k2 * (L * 1e-3) * (L * 1e-3));
    return A * exp(-1.0 / (k2 * L * L)) / k4 * kw * kw * damp;
}
√P(k) fftshift · sqrt 拉伸
两瓣方向沿风向 ŵ。L 越大瓣越向中心收缩（保留更多高频），V 越大整体能量越强。
03 初始复振幅 H₀(k) = ξ · √(P/2) FREQ
把噪声 ξr+i·ξi（复数）按 √(P/2) 缩放，得到每个 bin 的初始复振幅。 √(P/2) 是实标量，所以乘法就是"两个分量各乘一遍"。1/√2 是把方差 2 归回 1。
H₀(k) = (ξᵣ + iξᵢ) · √(P(k) / 2)   // 满足 E[|H₀|²] = P(k)
HLSL · 生成 H₀ 
RWTexture2D<float2> H0;             // 输出复数 (Re, Im)
Texture2D<float4>   GaussianTex;

[numthreads(8, 8, 1)]
void CS_GenerateH0(uint3 id : SV_DispatchThreadID) {
    // 标准 FFT 布局：DC 在角落，负频卷绕到后半
    int n = (id.x < N/2) ? (int)id.x : (int)id.x - (int)N;
    int m = (id.y < N/2) ? (int)id.y : (int)id.y - (int)N;
    float2 k = float2(n, m) * (6.2831853 / L_phys);

    float2 xi  = GaussianTex.Load(int3(id.xy, 0)).xy;
    float  amp = sqrt(Phillips(k) * 0.5);
    H0[id.xy] = xi * amp;                       // 复数 × 实标量 → 分量缩放
}
|H₀(k)| fftshift · sqrt 拉伸
04 时变频谱 H(k,t)（深水色散 + 共轭对称） ANIM
每帧只对 H₀ 乘相位旋转因子 e^{iωt}， 不需要重抽噪声。共轭对子 H₀*(-k)·e^{-iωt} 保证 IFFT 输出是实数。

关键：step 1 的噪声 ξ 是一次烘焙永不变， 真正"每帧动起来"的是 H(k,t) 的实/虚分量（每个 bin 以自己的 ω 在复平面上转圈）。 Shadertoy 里看到的"相位/phase"预览，就是下面 Re[H]、Im[H] 这两张图。 |H| 的模几乎不变（两共轭波只产生慢的干涉拍频），但 Re/Im 旋转得很快。
ω(k) = √(g · |k|)   // g ≈ 9.81 m/s²
H(k, t) = H₀(k) · e-iωt + H₀*(-k) · e+iωt   // 注意是 e^{−iωt}（顺风传播）
HLSL · 每帧更新频谱 
Texture2D<float2>   H0;
RWTexture2D<float2> H;
cbuffer Frame { uint N; float L_phys; float Time; };

float2 cmul(float2 a, float2 b) {
    return float2(a.x*b.x - a.y*b.y, a.x*b.y + a.y*b.x);
}

[numthreads(8, 8, 1)]
void CS_UpdateSpectrum(uint3 id : SV_DispatchThreadID) {
    int n = (id.x < N/2) ? (int)id.x : (int)id.x - (int)N;
    int m = (id.y < N/2) ? (int)id.y : (int)id.y - (int)N;
    float2 k = float2(n, m) * (6.2831853 / L_phys);
    float  w = sqrt(9.81 * length(k));

    uint2  mirror = (uint2(N, N) - id.xy) % N;
    float2 h_pk   = H0[id.xy];
    float2 h_mk   = H0[mirror];  h_mk.y = -h_mk.y;     // 共轭

    float c = cos(w * Time), s = sin(w * Time);
    // 注意 sign：用 e^{-iωt}（不是 Tessendorf 原版 e^{+iωt}）
    // 配合 IFFT 的 e^{+ikx} 约定 → 波朝 +k 方向（顺风）流动
    H[id.xy] = cmul(h_pk, float2(c, -s))                // · e^{-iωt}
             + cmul(h_mk, float2(c,  s));                // · e^{+iωt}
}
|H(k,t)| 模 · 慢变（干涉拍频）
Re[H(k,t)] 实部 · 快变（相位旋转）
Im[H(k,t)] 虚部 · 快变（相位旋转）
Re/Im 两张图就是 Shadertoy "phase / complex" 预览的真身：每个 bin 以 ω(k)=√(g|k|) 的角速度在复平面转圈，所以高频区像沙暴一样快速翻腾，低频区缓慢漂动。
05 2D IFFT → 高度场 h(x,t) IFFT
把频谱 H(k,t) 经 2D IFFT 转回空间域，取实部就是水面高度（虚部理论为 0，可丢）。 GPU 实际跑共享内存版 Cooley-Tukey，行/列各一次。
h(x, t) = Re[IFFT2D{ H(k, t) }]
HLSL · 行方向 Cooley-Tukey IFFT 
#define N 256
groupshared float2 g[N];

Texture2D<float2>   Src;
RWTexture2D<float2> Dst;

[numthreads(N, 1, 1)]
void CS_IFFT_Row(uint3 id : SV_DispatchThreadID,
                 uint  row : SV_GroupID) {
    // 1) bit-reversal 载入
    uint rev = reversebits(id.x) >> (32 - 8);
    g[id.x]  = Src[uint2(rev, row)];
    GroupMemoryBarrierWithGroupSync();

    // 2) log₂N 阶蝶形（IFFT 用 +π）
    [unroll] for (uint s = 1; s < N; s <<= 1) {
        if ((id.x & s) == 0) {
            float  ang = 3.1415927 * (id.x & (s - 1)) / s;
            float2 tw  = float2(cos(ang), sin(ang));
            float2 a   = g[id.x];
            float2 b   = cmul(g[id.x + s], tw);
            g[id.x]     = a + b;
            g[id.x + s] = a - b;
        }
        GroupMemoryBarrierWithGroupSync();
    }

    Dst[uint2(id.x, row)] = g[id.x] * (1.0 / N);
}
// 列方向：换坐标轴或先转置，复用同一 shader
h(x, t) 最终高度场（灰阶）
06 频域梯度 → 斜率 ∂h/∂x、∂h/∂z IFFT × 2
"频域乘 i·k = 空间域求偏导"。对 H 各乘 i·kx 与 i·kz 后再 IFFT，分别得到两个方向的斜率。 两次额外 IFFT，但比空间域差分精度高得多（频域微分零误差）。
∂h/∂x = Re[IFFT{ i kx · H(k) }]
∂h/∂z = Re[IFFT{ i kz · H(k) }]
HLSL · 频域预乘 i·k 
Texture2D<float2>   H;
RWTexture2D<float2> SlopeXFreq, SlopeZFreq;

[numthreads(8, 8, 1)]
void CS_PreSlope(uint3 id : SV_DispatchThreadID) {
    int n = (id.x < N/2) ? (int)id.x : (int)id.x - (int)N;
    int m = (id.y < N/2) ? (int)id.y : (int)id.y - (int)N;
    float2 k = float2(n, m) * (6.2831853 / L_phys);
    float2 h = H[id.xy];

    // i·k·(a + ib) = -k·b + i·k·a
    SlopeXFreq[id.xy] = float2(-k.x * h.y, k.x * h.x);
    SlopeZFreq[id.xy] = float2(-k.y * h.y, k.y * h.x);
}
// 然后对两张图各做一次 IFFT，取实部 = ∂h/∂x、∂h/∂z
∂h/∂x 水平方向斜率
∂h/∂z 深度方向斜率
07 单位法线 → 法线贴图 (RGB) PACK
由两个方向斜率拼出几何法线，归一化后编码到 [0,1] 写入 RGB。 这正是 PBR 海面材质用来打灯的法线贴图。
N = normalize( -∂h/∂x, 1, -∂h/∂z)
RGB = N * 0.5 + 0.5
HLSL · PS 重建法线 
Texture2D<float> SlopeXTex, SlopeZTex;
SamplerState     samLinear;

float3 PSOceanNormal(float2 uv : TEXCOORD0) : SV_TARGET {
    float  sx = SlopeXTex.Sample(samLinear, uv);
    float  sz = SlopeZTex.Sample(samLinear, uv);
    float3 N  = normalize(float3(-sx, 1.0, -sz));
    return N * 0.5 + 0.5;
}
Normal RGB R=Nx, G=Nz, B=Ny
08 Choppy 水平位移 Dx, Dz（FFT 版 Gerstner） IFFT × 2
把每个顶点沿波传播方向水平挤一下，得到尖锐波峰 + 平坦波谷。乘 -i·k/|k| 后再 IFFT；λ 是 choppy 强度（>1.5 会折叠）。
Dx = Re[IFFT{ -i · kx / |k| · H }]
Dz = Re[IFFT{ -i · kz / |k| · H }]
P = (x + λDx, h, z + λDz)
HLSL · 频域预乘 −i·k/|k| + VS 应用位移 
RWTexture2D<float2> DispXFreq, DispZFreq;

[numthreads(8, 8, 1)]
void CS_PreChoppy(uint3 id : SV_DispatchThreadID) {
    int n = (id.x < N/2) ? (int)id.x : (int)id.x - (int)N;
    int m = (id.y < N/2) ? (int)id.y : (int)id.y - (int)N;
    float2 k    = float2(n, m) * (6.2831853 / L_phys);
    float  kLen = length(k);
    float2 h    = H[id.xy];
    float2 fx   = kLen > 1e-6 ? float2(0, -k.x / kLen) : 0;
    float2 fz   = kLen > 1e-6 ? float2(0, -k.y / kLen) : 0;
    DispXFreq[id.xy] = cmul(h, fx);
    DispZFreq[id.xy] = cmul(h, fz);
}

// 顶点 shader 应用
VSOut VSOcean(float3 pos : POSITION, float2 uv : TEXCOORD0) {
    float  h  = HeightTex.SampleLevel(samLinear, uv, 0);
    float2 D  = float2(DispXTex.SampleLevel(samLinear, uv, 0),
                       DispZTex.SampleLevel(samLinear, uv, 0));
    pos.xz += lambda * D;
    pos.y  += h;
    return Project(pos, uv);
}
Dx x 方向水平位移
Dz z 方向水平位移
λ 滑条控制 choppy 强度。0=纯正弦，1=Tessendorf 推荐，>1.5 出现折叠（mesh 自交）。
09 最终合成 · 着色海面预览 ANIM
把高度 + 法线 + Choppy 位移合在一起着色：基色按深浅渐变、Lambert 漫反射、Schlick Fresnel、 Blinn-Phong 高光。这就是 Tessendorf 流程落到屏幕上的最终样子。
N = normalize(-∂h/∂x, 1, -∂h/∂z)
diff = max(0, dot(N, L))
fres = (1 - dot(N, V))5
color = lerp(deep, shallow, h̃) · diff + sky · fres + sun · spec
HLSL · 海面着色 PS 
float4 PSOceanShade(VSOut i) : SV_TARGET {
    float  h  = HeightTex.Sample(samLinear, i.uv);
    float  sx = SlopeXTex.Sample(samLinear, i.uv);
    float  sz = SlopeZTex.Sample(samLinear, i.uv);
    float3 N  = normalize(float3(-sx, 1.0, -sz));

    float  diff = saturate(dot(N, lightDir));
    float  NdotV = saturate(dot(N, viewDir));
    float  fres = pow(1.0 - NdotV, 5.0);
    float3 H    = normalize(lightDir + viewDir);
    float  spec = pow(saturate(dot(N, H)), 64.0);

    float3 base = lerp(deepColor, shallowColor, saturate(h * 0.5 + 0.5));
    float3 col  = base * (0.3 + 0.7 * diff)
                + skyColor * fres * 0.5
                + sunColor * spec;
    return float4(col, 1.0);
}
Shaded Ocean 高度 + 法线 + Choppy 位移 + 光照
每帧 5 次 IFFT（h, ∂x, ∂z, Dx, Dz）+ 一次着色，就是 Tessendorf FFT 海面的完整运行时开销。
10 单层 → Cascade · 多尺度演进 SCALE
单张 FFT patch 受 N 和 L 共同约束，能解析的波长跨度只有 4·dx ~ L/2（约 5 octave）。 真实海洋从 cm 级毛细波到 km 级涌浪有 14 octave 跨度 —— 必须用 Cascade： 把 step 1–9 的整套管线复制 N 份，每份用不同 L_phys，各自负责一段波长。 页面顶部的 World View 就是 4 cascade 的实战演示，下面是从 single 到 cascade 的 6 个具体变更点。
// (A) 配置：N 个 cascade，L 取互质数避免周期同步
const Cascades = [
  { name: "Detail", L: 27, fadeFar: 28 },
  { name: "Near", L: 211, fadeFar: 250 },
  { name: "Mid", L: 1583, fadeFar: 2500 },
  { name: "Far", L: 11873, fadeFar: Infinity },
]; // 27·211·1583·11873 都是质数，叠加后无可见 tile 周期
// (B) 状态：每 cascade 复制一份完整 step 1–9 的数据
struct CascadeState {
  noiseR, noiseI   // step 1: 一次烘焙
  phillips         // step 2: 参数变才重算
  h0Re, h0Im     // step 3: 同上
  hRe, hIm       // step 4: 每帧重算
  height, slopeX, slopeZ, dispX, dispZ  // step 5–8: 每帧 5 次 IFFT
}
// (C) 带通过滤：每 cascade 只贡献自己负责的波长段
λ = 2π / |k|
mask(λ) = smoothstep(λ_min/2, λ_min, λ) · (1 - smoothstep(λ_max, 2λ_max, λ))
P_cascade(k) = Phillips(k) · mask(λ)  // 相邻 cascade 不再贡献同一波长，避免双重计算
// (D) RMS 归一化：让每 cascade 视觉幅度对等
RMS_slope = √(mean(slopeX2 + slopeZ2) / 2)
slope_norm = slope / RMS_slope  // Detail 短波本来振幅极小，归一化后能盖过 Far 巨振幅
// (E) 世界空间采样 + 周期 wrap
uv = worldXZ / L_cascade
uv = uv - floor(uv)  // IFFT 输出天然周期，wrap 无缝
// (F) 距离 fade（按到观察者的距离，不是相机）
dist = |worldXZ - observerXZ|
w_i = 1 - smoothstep(fadeNear_i, fadeFar_i, dist)
final = Σ_i w_i · slope_i_norm(uv_i)
HLSL · Phillips 函数 加带通（替换原 step 1 末尾） 
float PhillipsCascade(float2 k, float bandMin, float bandMax) {
    float k2 = dot(k, k);
    if (k2 < 1e-12) return 0.0;
    float L  = V * V / 9.81;
    float kw = max(0.0, dot(normalize(k), windDir));
    float damp = exp(-k2 * (L * 1e-3) * (L * 1e-3));
    float P = A * exp(-1.0 / (k2 * L * L)) / (k2 * k2) * kw * kw * damp;

    // === 新增：带通 mask ===
    float lambda = 6.2831853 / sqrt(k2);
    float loFade = smoothstep(bandMin * 0.5, bandMin, lambda);
    float hiFade = 1.0 - smoothstep(bandMax, bandMax * 2.0, lambda);
    P *= loFade * hiFade;

    return P;
}
HLSL · CPU 端预算 RMS（每次 IFFT 完后做一次） 
// 每个 cascade 跑完 5× IFFT 后，CPU 算 RMS slope/height
void UpdateCascadeRMS(Cascade& c) {
    float sumSqSlope = 0, sumSqHeight = 0;
    int total = c.N * c.N;
    for (int i = 0; i < total; ++i) {
        sumSqSlope  += c.slopeX[i]*c.slopeX[i] + c.slopeZ[i]*c.slopeZ[i];
        sumSqHeight += c.height[i] * c.height[i];
    }
    c.invRmsSlope  = 1.0 / sqrt(sumSqSlope / (2.0 * total));
    c.invRmsHeight = 1.0 / sqrt(sumSqHeight / total);
    // 上传到 GPU 常量缓冲
}

cbuffer CascadeParams {

    float  fadeFar     [4];
    float  invRmsSlope [4];   // 预算的 1/RMS
    float  invRmsHeight[4];
    float2 observerXZ;        // 观察者世界位置（不是相机！）
};

struct OceanSample { float h; float2 slope; };

OceanSample SampleOcean(float2 worldXZ) {
    float dist = distance(worldXZ, observerXZ);    // ← 到观察者，不是相机

    OceanSample o = (OceanSample)0;
    [unroll] for (uint i = 0; i < 4; ++i) {
        float w = 1.0 - smoothstep(fadeNear[i], fadeFar[i], dist);
        if (w < 0.001) continue;

        float2 uv = worldXZ / L_cascade[i];          // 世界空间 → cascade UV（自动 wrap）
        o.h     += w * invRmsHeight[i] * HeightTex[i].Sample(samLinear, uv);
        o.slope += w * invRmsSlope[i]  * SlopeTex [i].Sample(samLinear, uv);
    }
    return o;
}

完整 cascade 系统 = 单层 × N 份独立管线 + 带通过滤 + RMS 归一化 + 距离 fade。 4 cascade × 5 IFFT/帧 = 每帧 20 个 IFFT，N=256 在 PS5 上约 1.5ms。 页面顶部演示：拖相机离观察者越远，内层 cascade（Detail/Near）按 fade 距离逐圈消失，剩 Mid/Far 撑画面 —— 这就是 LOD 的本质。
常见坑： ① L 不互质 → 叠加后出现可见周期 tile pattern； ② 不带通 → 相邻 cascade 重复贡献，能量翻倍； ③ 不归一化 → Far 振幅压死 Detail； ④ fade 用相机距离而非观察者 → 相机一动 LOD 边界跟着抖（应该锚定在观察者/角色身上）。
