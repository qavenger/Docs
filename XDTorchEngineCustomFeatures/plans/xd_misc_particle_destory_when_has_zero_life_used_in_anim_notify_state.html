bool UParticleSystem::IsAnyEmitterHasInfiniteLifetime()
{
#if XD_MISC_PARTICLE_DESTORY_WHEN_HAS_ZERO_LIFE_USED_IN_ANIM_NOTIFY_STATE
#if !UE_EDITOR
	if (bAlreadyCheckAnyEmitterHasInfiniteLifetime)
	{
		return bHasAnyEmitterHasInfiniteLifetime;
	}
#endif
	for (auto& Emitter : Emitters)
	{
		if (!Emitter) { continue; }   // 必须保留：Cook 平台排除会把 Emitters[i] 置为 nullptr (ParticleComponents.cpp:3037)
		for (auto& LodLevel : Emitter->LODLevels)
		{
			if (LodLevel && LodLevel->bEnabled && LodLevel->RequiredModule && !LodLevel->RequiredModule->bKillOnDeactivate)
			{
				bool InfiniteLifetime = false;
				bool bFoundLifetimeModule = false;
				for (auto& ParticleModule : LodLevel->Modules)
				{
					if (auto* LifetimeModule = Cast<UParticleModuleLifetimeBase>(ParticleModule))
					{
						if (LifetimeModule->GetMaxLifetime() <= 0.0f) { InfiniteLifetime = true; }
						bFoundLifetimeModule = true;
						break;
					}
				}
				if (!bFoundLifetimeModule) { InfiniteLifetime = true; }
				if (InfiniteLifetime)
				{
#if !UE_EDITOR
					bAlreadyCheckAnyEmitterHasInfiniteLifetime = true;
					bHasAnyEmitterHasInfiniteLifetime = true;
#endif
					return true;
				}
			}
		}
	}
#if !UE_EDITOR
	bAlreadyCheckAnyEmitterHasInfiniteLifetime = true;
	bHasAnyEmitterHasInfiniteLifetime = false;
#endif
	return false;
#else
	return false;
#endif
}
// 逐字对照 UE4 Particles/ParticleComponents.cpp:3795-3858；声明加在 UE5 ParticleSystem.h 的 public 段（UE4 对应 ParticleSystem.h:693-706）
// 同源码加在 UE5 ParticleSystem.cpp；头文件加入 ParticleSystem.h 的 public 段
UFUNCTION(BlueprintCallable, Category="Particle System")
bool IsAnyEmitterHasInfiniteLifetime();

⚠ UE5 LODLevel 仍为 TArray<TObjectPtr<UParticleLODLevel>>，RequiredModule/bKillOnDeactivate 字段名未变，直接可用；缓存字段可选。

风险评估

原代码风险：每次调用遍历所有 emitter/LOD/module，虽非热路径但若在 NotifyBegin 每帧触发会有开销（UE4 用 bAlreadyCheck 缓存缓解）。

移植风险：改的是公共 AnimNotify（所有项目复用），B 类资产行为会变；UE5 Notify 签名已变，漏改 EventReference 会编译失败。

开发自检

  • 含 constant 0 寿命 emitter 的粒子在 NotifyState 结束时组件被 Destroy
  • 有限寿命粒子仍走 DeactivateSystem 正常播完

结论置信度：medium — UE5 AnimNotifyState_TimedParticleEffect.h 是否已有可复用的判定/开关需再确认；bKillOnDeactivate 语义是否与 UE4 完全一致。
